        floor = _aggregatePrincipalFloor(
            P, v1Staked, _maxLossBps(_strategyFor(t)), ICutoverStaker(address(v2)).stakerCount(t), WEI_SLACK
        );
    }

    function resetTokens() external {
        delete tokens;
    }

    function _tryGlobalPause(string memory tag) internal returns (bool ok, bytes memory err) {
        address actor = makeAddr("run33-eye-actor");

    function _reason(bytes memory ret) internal pure returns (string memory) {
        if (ret.length < 68) return string(abi.encodePacked("raw:", ret));
        bytes memory body = new bytes(ret.length - 4);
        for (uint256 i = 0; i < body.length; i++) {
            body[i] = ret[i + 4];
        }
        return abi.decode(body, (string));
    }

    function _contains(string memory hay, string memory needle) internal pure returns (bool) {
        bytes memory a = bytes(hay);
        bytes memory b = bytes(needle);
        if (b.length == 0 || b.length > a.length) return false;
        for (uint256 i = 0; i <= a.length - b.length; i++) {
            bool m = true;
            for (uint256 j = 0; j < b.length; j++) {
                if (a[i + j] != b[j]) {
                    m = false;
                    break;
                }
            }
            if (m) return true;
        }
        return false;
    }

    // ======================================================================================================
    // F-01 falsification: the story-085 floor can fail where the old lockstep cannot
    // ======================================================================================================
    function _haircutDolaThenMigrate(uint256 haircutBps)
        internal
        returns (address[] memory users, uint256[] memory pre, uint256[] memory credit)
    {
        _through5();
        h.p6Setup();
        address vault = ICutoverStrategy(YS_DOLA).vault();
        uint256 shares = IERC20(vault).balanceOf(YS_DOLA);
        if (haircutBps > 0) {
            // Shares are commingled across the strategy's clients: leave the strategy holding only
            // (1 - haircut) of the shares V1's principal needs, so V1's exit realizes R ~= P * (1 - haircut).
            (,,, uint256 staked) = ICutoverStaker(V1).poolInfo(DOLA);
            uint256 need = I4626Like(vault).convertToShares(staked);
            uint256 keep = need * (10_000 - haircutBps) / 10_000;
            require(shares > keep, "harness: strategy holds fewer shares than the haircut target");
            vm.prank(YS_DOLA);
            IERC20(vault).transfer(makeAddr("haircut-sink"), shares - keep);
        }
        h.initiate(DOLA);
        (uint256 R, uint256 P) = ICutoverStaker(V1).migrationInfo(DOLA);
        console2.log(string.concat("HAIRCUT|bps=", vm.toString(haircutBps), "|R=", vm.toString(R), "|P=", vm.toString(P)));
        (users, pre, credit) = h.migrateNoAssert(DOLA);
    }

    // ======================================================================================================
    // NEW: a V1 staker who self-exits (permissionless userMigrate) between the initiateMigration tx and the
    // migrate tx leaves a CORRECT cutover that the aggregate floor (verifier + any resume/preview leg) rejects.
    // ======================================================================================================
    function _cutoverWithOptionalSelfExit(bool selfExit) internal returns (Vm.Log[] memory logs, address exiter, uint256 exitAmt) {
        _through5();
        vm.recordLogs();
        h.p6Setup();
        // DOLA: exactly the broadcast shape - list planned in the local pass, a self-exit lands between txs.
        h.initiate(DOLA);
        address[] memory planned = h.planMigratable(DOLA);
        if (selfExit) {
            for (uint256 i = 0; i < planned.length; i++) {
                (uint256 a,) = ICutoverStaker(V1).userInfo(DOLA, planned[i]);
                if (a > exitAmt) {
                    exitAmt = a;
                    exiter = planned[i];
                }
            }
            // choose the SMALLEST staker above 1 DOLA to show the threshold is low
            for (uint256 i = 0; i < planned.length; i++) {
                (uint256 a,) = ICutoverStaker(V1).userInfo(DOLA, planned[i]);
                if (a > 1e18 && a < exitAmt) {
                    exitAmt = a;
                    exiter = planned[i];
                }
            }
            uint256 balBefore = IERC20(DOLA).balanceOf(exiter);
            vm.prank(exiter);
            IV1User(V1).userMigrate(DOLA);
            console2.log(string.concat(
                "SELF_EXIT|user=", vm.toString(exiter), "|V1principal=", vm.toString(exitAmt),
                "|paidToWallet=", vm.toString(IERC20(DOLA).balanceOf(exiter) - balBefore)
            ));
        }
        h.rawMigrate(DOLA, planned); // on chain: already-exited user skipped by batchMigrate
        h.migrateAndAssert(USDC);
        h.migrateAndAssert(USDE);
        h.p7();
        logs = vm.getRecordedLogs();
        h.p8(); // real Phase 8 wiring assertions pass
        assertEq(ICutoverStaker(V1).stakerCount(DOLA), 0, "every DOLA V1 staker exited (migrated or self-exited)");
    }

    /// Resume consequence: broadcast halts after the DOLA migrate (before USDC/USDe and Phase 7) and a DOLA staker had
    /// self-exited in between. The resume leg (same local pass as :broadcast / :preview) reverts at DOLA's Phase-6 floor
    /// before it can migrate USDC/USDe or finalize - the cutover cannot be completed without editing the script.
    function test_SX_haltedResumeCannotFinish() public {
        _through5();
        h.p6Setup();
        h.initiate(DOLA);
        address[] memory planned = h.planMigratable(DOLA);
        address exiter;
        uint256 best = type(uint256).max;
        for (uint256 i = 0; i < planned.length; i++) {
            (uint256 a,) = ICutoverStaker(V1).userInfo(DOLA, planned[i]);
            if (a > 1e18 && a < best) {
                best = a;
                exiter = planned[i];
            }
        }
        vm.prank(exiter);
        IV1User(V1).userMigrate(DOLA);
        h.rawMigrate(DOLA, planned);
        // ---- HALT. Resume: ----
        h.resetTokens();
        try h.run() {
            revert("resume unexpectedly completed");
        } catch (bytes memory ret) {
            string memory r = _reason(ret);
            console2.log(string.concat("SX_RESUME|exiterPrincipal=", vm.toString(best), "|", r));
            assertTrue(_contains(r, "V2 booked total below pre-migration principal floor"));
        }
        assertEq(ICutoverStaker(V1).poolState(USDC), 0, "USDC pool never initiated");
        assertTrue(IPausableLike(address(h.v2())).paused(), "V2 left paused");
        assertFalse(IPauserRegistry(PAUSER).isRegistered(address(h.v2())), "V2 not under the global breaker");
    }
